cmake_minimum_required(VERSION 3.22)
project(vc_reader_extensions LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
option(VC_READER_BUILD_PYTHON "Build Python extensions" ON)
option(VC_READER_BUILD_READER "Build Zarr reading and sampling APIs" ON)
option(VC_READER_BUILD_NATIVE_TESTS "Build native C++ API tests" OFF)

# Initialize CUDA architecture defaults before creating targets.
# An explicit CPU build skips CUDA discovery entirely.
if(NOT DEFINED VC_READER_ENABLE_CUDA OR VC_READER_ENABLE_CUDA)
    if(NOT DEFINED CMAKE_CUDA_COMPILER AND NOT DEFINED ENV{CUDACXX})
        # Toolkit discovery also searches standard locations outside PATH.
        find_package(CUDAToolkit QUIET)
        if(CUDAToolkit_NVCC_EXECUTABLE)
            set(CMAKE_CUDA_COMPILER "${CUDAToolkit_NVCC_EXECUTABLE}"
                CACHE FILEPATH "CUDA compiler")
        endif()
    endif()
    include(CheckLanguage)
    check_language(CUDA)
endif()
set(_vc_reader_cuda_default OFF)
if(CMAKE_CUDA_COMPILER)
    set(_vc_reader_cuda_default ON)
endif()
option(VC_READER_ENABLE_CUDA "Build optional CUDA reader and transform backends"
    ${_vc_reader_cuda_default})
if(VC_READER_ENABLE_CUDA)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
endif()


include(cmake/NativeTargets.cmake)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(CURL REQUIRED)
find_package(ZLIB REQUIRED)
add_library(vc_reader_catalog STATIC src/cpp/catalog.cpp src/cpp/catalog_io.cpp src/cpp/registrations.cpp)
add_library(vc_reader::catalog ALIAS vc_reader_catalog)
set_target_properties(vc_reader_catalog PROPERTIES EXPORT_NAME catalog POSITION_INDEPENDENT_CODE ON AUTOMOC OFF AUTOUIC OFF)
target_link_libraries(vc_reader_catalog PUBLIC vc_reader::headers nlohmann_json::nlohmann_json PRIVATE CURL::libcurl ZLIB::ZLIB)
target_compile_features(vc_reader_catalog PUBLIC cxx_std_20)
if(VC_READER_BUILD_READER)
    include(cmake/NativeReader.cmake)
endif()
include(cmake/InstallNative.cmake)

if(VC_READER_BUILD_NATIVE_TESTS)
    enable_testing()
    add_executable(test_native_catalog tests/cpp/test_native_catalog.cpp)
    target_link_libraries(test_native_catalog PRIVATE vc_reader::catalog)
    add_test(NAME test_native_catalog COMMAND test_native_catalog "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/training/catalog.json")
    add_executable(test_native_transforms tests/cpp/test_native_transforms.cpp)
    target_link_libraries(test_native_transforms PRIVATE vc_reader::transforms)
    add_test(NAME test_native_transforms COMMAND test_native_transforms)
    if(VC_READER_BUILD_READER)
        add_executable(test_native_reader tests/cpp/test_native_reader.cpp)
        target_link_libraries(test_native_reader PRIVATE vc_reader::reader)
        add_test(NAME test_native_reader COMMAND test_native_reader)
        add_executable(test_native_sampling tests/cpp/test_native_sampling.cpp)
        target_link_libraries(test_native_sampling PRIVATE vc_reader::reader)
        add_test(NAME test_native_sampling COMMAND test_native_sampling)
    endif()
    if(VC_READER_ENABLE_CUDA)
        add_executable(test_native_cuda tests/cpp/test_native_cuda.cpp)
        target_link_libraries(test_native_cuda PRIVATE vc_reader::transforms)
        if(VC_READER_BUILD_READER)
            target_link_libraries(test_native_cuda PRIVATE vc_reader::reader)
            target_compile_definitions(test_native_cuda PRIVATE VCR_TEST_READER=1)
        endif()
        add_test(NAME test_native_cuda COMMAND test_native_cuda)
        set_tests_properties(test_native_cuda PROPERTIES SKIP_RETURN_CODE 77)
    endif()
endif()

option(VC_READER_BUILD_EXAMPLES "Build native C++ examples" OFF)
if(VC_READER_BUILD_EXAMPLES AND VC_READER_BUILD_READER)
    add_executable(vc_reader_ct_normals examples/ct_normals.cpp)
    target_link_libraries(vc_reader_ct_normals PRIVATE vc_reader::reader)
endif()

if(NOT VC_READER_BUILD_PYTHON)
    return()
endif()
find_package(Python 3.10 COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir
    OUTPUT_VARIABLE nanobind_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(nanobind CONFIG REQUIRED)
if(nanobind_VERSION VERSION_LESS 2.10.2)
    message(FATAL_ERROR "vc_reader requires nanobind >= 2.10.2 for device-array ownership")
endif()
nanobind_add_module(_catalog NB_STATIC NOMINSIZE src/cpp/catalog_bindings.cpp)
target_link_libraries(_catalog PRIVATE vc_reader::catalog)
install(TARGETS _catalog LIBRARY DESTINATION vc_reader)
# Python and native clients use the same implementation and CUDA libraries.
nanobind_add_module(_transforms NB_STATIC NOMINSIZE src/cpp/transforms/bindings.cpp)
target_link_libraries(_transforms PRIVATE vc_reader::transforms)
install(TARGETS _transforms LIBRARY DESTINATION vc_reader)
if(VC_READER_BUILD_READER)
    nanobind_add_module(_reader NB_STATIC NOMINSIZE src/cpp/bindings.cpp)
    target_link_libraries(_reader PRIVATE vc_reader::reader)
    if(APPLE)
        set_target_properties(_reader PROPERTIES INSTALL_RPATH "@loader_path/../lib")
    else()
        set_target_properties(_reader PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
    endif()
    install(TARGETS _reader LIBRARY DESTINATION vc_reader)
endif()
