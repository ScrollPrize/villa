cmake_minimum_required(VERSION 3.22)
project(test_delta3d_embedding LANGUAGES C CXX)
include(FetchContent)
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/FetchDelta3D.cmake")
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
set(VC_DELTA3D_BUILD_PYTHON OFF)
set(VC_DELTA3D_BUILD_TESTS OFF)
set(VC_DELTA3D_INSTALL_CPP OFF)
set(VC_DELTA3D_MODERN_SMOKE ON)
FetchContent_Declare(vc_delta3d
    GIT_REPOSITORY https://github.com/ScrollPrize/vc-delta3d.git
    GIT_TAG 45339743b450ff707d21fb742dc1b3b2bf99306d)
vc_fetch_delta3d()
foreach(target vc_delta3d vc_ng_fixed32 vc_ng_fixed64 vc_ng_fixed128 vc_ng_fixed256)
    foreach(property AUTOMOC AUTOUIC AUTORCC)
        get_target_property(value ${target} ${property})
        if(value)
            message(FATAL_ERROR "${target} inherited ${property}")
        endif()
    endforeach()
endforeach()
if(NOT CMAKE_AUTOMOC OR NOT CMAKE_AUTOUIC OR NOT CMAKE_AUTORCC)
    message(FATAL_ERROR "Codec fetch changed the parent's Qt settings")
endif()
enable_testing()
add_test(NAME delta3d_embedding COMMAND vc_delta3d_modern_smoke)
