cmake_minimum_required(VERSION 3.22)
project(vc_object_store VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
if(APPLE)
    set(CMAKE_INSTALL_RPATH "@loader_path/../lib")
else()
    set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(VC_OBJECT_STORE_BUILD_ADAPTERS "Build native formats and relationship detectors" ON)
option(VC_OBJECT_STORE_BUILD_TOOLS "Build command-line tools" ON)
option(VC_OBJECT_STORE_BUILD_PYTHON "Build Python package" OFF)
option(VC_OBJECT_STORE_BUILD_TESTS "Build tests and benchmarks" OFF)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ZSTD REQUIRED IMPORTED_TARGET libzstd)
add_library(zstd::libzstd ALIAS PkgConfig::ZSTD)
if(VC_OBJECT_STORE_BUILD_ADAPTERS)
    find_package(OpenCV REQUIRED COMPONENTS core)
    find_package(OpenMP REQUIRED)
endif()
add_subdirectory(src)
if(VC_OBJECT_STORE_BUILD_TOOLS)
    if(NOT VC_OBJECT_STORE_BUILD_ADAPTERS)
        message(FATAL_ERROR "Tools require adapters")
    endif()
    add_executable(vc_object_ingest apps/vc_object_ingest.cpp)
    add_executable(vc_object_export apps/vc_object_export.cpp)
    add_executable(vc_object_links_cli apps/vc_object_links.cpp)
    set_target_properties(vc_object_links_cli PROPERTIES OUTPUT_NAME vc_object_links)
    target_link_libraries(vc_object_ingest PRIVATE vc_object_geometry)
    target_link_libraries(vc_object_export PRIVATE vc_object_geometry)
    target_link_libraries(vc_object_links_cli PRIVATE vc_object_links nlohmann_json::nlohmann_json)
    install(TARGETS vc_object_ingest vc_object_export vc_object_links_cli RUNTIME DESTINATION bin)
endif()
if(VC_OBJECT_STORE_BUILD_PYTHON)
    if(NOT VC_OBJECT_STORE_BUILD_ADAPTERS)
        message(FATAL_ERROR "Python package requires adapters")
    endif()
    add_subdirectory(python)
endif()
if(VC_OBJECT_STORE_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests/cpp)
endif()

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(_vc_store_targets vc_geometry vc_objects)
if(VC_OBJECT_STORE_BUILD_ADAPTERS)
    list(APPEND _vc_store_targets vc_object_geometry vc_object_links vc_object_formats vc_track_crossings)
endif()
install(TARGETS ${_vc_store_targets} EXPORT vc_object_storeTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT vc_object_storeTargets NAMESPACE vc_object_store:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/vc_object_store)
configure_package_config_file(cmake/vc_object_storeConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/vc_object_storeConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/vc_object_store)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/vc_object_storeConfig.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/vc_object_store)
